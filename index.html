let students = []; async function loadStudents() { const tbody = document.getElementById("studentRows"); tbody.replaceChildren(); const message = document.createElement("tr"); const cell = document.createElement("td"); cell.colSpan = 4; cell.textContent = "Loading student records..."; message.appendChild(cell); tbody.appendChild(message); const { data, error } = await supabaseClient .from("students") .select("id, student_number") .limit(50); if (error) { cell.textContent = "Unable to load students: " + error.message; return; } students = data || []; renderStudents(); } function renderStudents(query = "") { const tbody = document.getElementById("studentRows"); tbody.replaceChildren(); const filtered = students.filter(student => String(student.student_number || "") .toLowerCase() .includes(query.toLowerCase()) ); if (filtered.length === 0) { const tr = document.createElement("tr"); const td = document.createElement("td"); td.colSpan = 4; td.textContent = "No accessible students found."; tr.appendChild(td); tbody.appendChild(tr); return; } filtered.forEach((student, index) => { const tr = document.createElement("tr"); [ "Student Record " + (index + 1), student.student_number || "—", "Not displayed yet", "Read-only" ].forEach(value => { const td = document.createElement("td"); td.textContent = value; tr.appendChild(td); }); tbody.appendChild(tr); }); }
